
import express from 'express';
import dotenv from 'dotenv';
import axios from 'axios';
import { parseStringPromise } from 'xml2js';

dotenv.config();
const app = express();
app.use(express.json());

const PROMPT_MAESTRO = `Eres el Contador Chingón de Puebla. SAT 2026. Habla corto, en español mexicano, sin tecnicismos. Ejemplo con dinero real siempre. RESICO 2026: hasta 25k=1% ISR, 25-50k=1.1%, 50-83k=1.5%, 83-250k=2%, 250-350k=2.5%. IVA 16%`;

async function enviarWhatsApp(to, text) {
  await axios.post(`https://graph.facebook.com/v19.0/${process.env.WHATSAPP_PHONE_ID}/messages`, {
    messaging_product: "whatsapp", to, type: "text", text: { body: text.slice(0,1000) }
  }, { headers: { Authorization: `Bearer ${process.env.WHATSAPP_TOKEN}` } });
}

// ESTA ES LA PARTE QUE COPIASTE, PERO EN NODE.JS - GRATIS CON GEMINI
async function preguntarGemini(pregunta) {
  const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-flash-latest:generateContent?key=${process.env.GEMINI_API_KEY}`;
  const res = await axios.post(url, {
    contents: [{ parts: [{ text: PROMPT_MAESTRO + "\nUsuario: " + pregunta }]}]
  });
  return res.data.candidates?.[0]?.content?.parts?.[0]?.text || "No entendí, ¿me lo repites?";
}

async function descargarMediaWhatsApp(mediaId) {
  const mediaInfo = await axios.get(`https://graph.facebook.com/v19.0/${mediaId}`, {
    headers: { Authorization: `Bearer ${process.env.WHATSAPP_TOKEN}` }
  });
  const file = await axios.get(mediaInfo.data.url, {
    headers: { Authorization: `Bearer ${process.env.WHATSAPP_TOKEN}` },
    responseType: 'text'
  });
  return file.data;
}

app.get('/webhook', (req, res) => {
  if (req.query['hub.verify_token'] === process.env.VERIFY_TOKEN) res.send(req.query['hub.challenge']);
  else res.sendStatus(403);
});

app.post('/webhook', async (req, res) => {
  try {
    const msg = req.body.entry?.[0]?.changes?.[0]?.value?.messages?.[0];
    if (!msg) return res.sendStatus(200);
    const from = msg.from;

    if (msg.type === 'document') {
      await enviarWhatsApp(from, "📄 Recibí tu factura XML, la estoy leyendo...");
      try {
        const xmlContent = await descargarMediaWhatsApp(msg.document.id);
        const parsed = await parseStringPromise(xmlContent);
        const comp = parsed['cfdi:Comprobante'] || parsed['Comprobante'];
        const total = comp?.$?.Total || '0';
        const subtotal = comp?.$?.SubTotal || '0';
        const iva = (parseFloat(total) - parseFloat(subtotal)).toFixed(2);
        await enviarWhatsApp(from, `✅ Factura leída:\n💰 Total: $${total}\n📊 Base: $${subtotal}\n🧾 IVA: $${iva}`);
      } catch(e) { await enviarWhatsApp(from, "No pude leer el XML, reenvíamelo."); }
      return res.sendStatus(200);
    }

    const texto = msg.text?.body || "Hola";
    const respuesta = await preguntarGemini(texto);
    await enviarWhatsApp(from, respuesta);

  } catch(e){ console.error(e.response?.data || e.message); }
  res.sendStatus(200);
});

app.get('/', (req,res) => res.send('Contador Chingón GRATIS ON - Puebla 2026 con Gemini Flash'));

app.listen(process.env.PORT || 3000, () => console.log('BOT GRATIS ON con gemini-flash-latest'));
